/* ================================================================
   UTILITY: Safe POST JSON (with timeout)
   ================================================================ */
async function postJson(url, body, timeoutMs = 20000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);
  try {
    const res = await fetch(url, {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify(body || {}),
      signal: controller.signal
    });
    const text = await res.text();
    if (!res.ok) throw new Error(text || res.statusText);
    return text ? JSON.parse(text) : {};
  } finally {
    clearTimeout(timer);
  }
}

const MatchingMode = { strict: "strict", relaxed: "relaxed", extended: "extended" };
const DEFAULT_MATCHING_MODE = MatchingMode.relaxed;

/* RELAXED MATCHING: Subject = Title OR Title appears in Subject OR Body */
function relaxedMatch(doc, email) {
  if (!doc || !email) return false;
  const title = (doc.Title || "").trim().toLowerCase();
  if (!title) return false;
  const subject = (email.Subject || "").toLowerCase();
  const body = (email.Body || "").toLowerCase();
  return subject === title || subject.includes(title) || body.includes(title);
}

/* Optional (not used directly, but available) */
function computeDocEmailPairs(docs, emails) {
  const pairs = [];
  for (const d of docs) {
    for (const e of emails) {
      if (relaxedMatch(d, e)) {
        pairs.push({ DocID: d.ID, EmailID: e.id || e.ID, Subject: e.Subject, Title: d.Title });
      }
    }
  }
  return pairs;
}

function groupBy(arr, keyFn) {
  const map = new Map();
  for (const item of arr || []) {
    const k = String(keyFn(item) ?? "");
    if (!map.has(k)) map.set(k, []);
    map.get(k).push(item);
  }
  return map;
}

/* Builders */
function buildCollections(raw) {
  const docs   = (raw?.data?.docs   || raw?.docs   || []).map(toDoc);
  const tasks  = (raw?.data?.tasks  || raw?.tasks  || []).map(toTask);
  const emails = (raw?.data?.emails || raw?.emails || []).map(toEmail);
  return { docs, tasks, emails };
}
function buildMatchedDocs(docs, emails) {
  const colMatchedDocs = [];
  const colDocEmailPairs = [];
  for (const d of docs) {
    const matches = emails.filter(e => relaxedMatch(d, e));
    if (matches.length > 0) {
      colMatchedDocs.push({
        ID: d.ID,
        Created: d.StartDate ?? d.Created ?? null,
        Title: d.Title,
        Category: d.Category,
        AssignmentStatus: d.Status,
        Status: d.Status,
        AssignedTo: d.AssignedTo,
        AttachmentLink: null,
        MatchingEmails: matches
      });
      for (const e of matches) {
        colDocEmailPairs.push({
          DocID: d.ID, DocTitle: d.Title, DocCategory: d.Category, DocStatus: d.Status,
          EmailSubject: e.Subject, EmailBody: e.Body, EmailFrom: e.From, EmailReceived: e.DateTimeReceived
        });
      }
    }
  }
  return { colMatchedDocs, colDocEmailPairs };
}
function buildDocTaskEmailMatrix(docs, tasks, emails) {
  const tasksByDoc = groupBy(tasks, t => t.RefIDD ?? "");
  const rows = [];
  for (const d of docs) {
    const relatedTasks  = tasksByDoc.get(String(d.ID)) || [];
    const relatedEmails = emails.filter(e => relaxedMatch(d, e));
    rows.push({
      Created: d.Created, ID: d.ID, Title: d.Title, Category: d.Category, Status: d.Status,
      AssignedTo: d.AssignedTo, SharePointLink: d.SharePointLink || null,
      Tasks: relatedTasks, Responses: relatedEmails,
      TaskCount: relatedTasks.length, ResponseCount: relatedEmails.length
    });
  }
  return rows;
}
function computeCounters(docs, tasks, emails, pairs, matched) {
  return { docs: docs.length, tasks: tasks.length, emails: emails.length, pairs: pairs.length, matched: matched.length };
}

  try {
    const parseStart = performance.now();
    const { docs, tasks, emails } = buildCollections(raw);
    logTelemetry(action, "Data_Parse_All", "Success", Math.round(performance.now() - parseStart),
      `Docs:${docs.length} Tasks:${tasks.length} Emails:${emails.length}`);

    const shapeStart = performance.now();
    const matrix = buildDocTaskEmailMatrix(docs, tasks, emails);
    logTelemetry(action, "Data_Shape_Matrix", "Success", Math.round(performance.now() - shapeStart),
      `Matrix rows:${matrix.length}`);

    const dashStart = performance.now();
    const { colMatchedDocs, colDocEmailPairs } = buildMatchedDocs(docs, emails);
    logTelemetry(action, "Dashboard_Recalculate", "Success", Math.round(performance.now() - dashStart),
      `Matched:${colMatchedDocs.length} Pairs:${colDocEmailPairs.length}`);

    const counts = computeCounters(docs, tasks, emails, colDocEmailPairs, colMatchedDocs);
    const total = Math.round(performance.now() - t0);
    logTelemetry(action, "Action_End", "Success", total, "Total execution time");

    return {
      counts,
      colDocsTracking: docs,
      colTasksTracking: tasks,
      colEmailsTracking: emails,
      colDocTaskEmailMatrix: matrix,
      colMatchedDocs, colDocEmailPairs,
      telemetry: window.__state.telemetry,
      user: raw?.user || window.__state.user,
      timestamp: raw?.timestamp || new Date().toISOString()
    };
  } catch (err) {
    const total = Math.round(performance.now() - t0);
    logTelemetry(action, "Action_End", "Error", total, String(err));
    return { error: err, telemetry: window.__state.telemetry };
  }
}

/* Sorting + Filtering helpers */
function sortDocs(rows, key) {
  const [field, dir] = key.split('_'); const mul = (dir === 'desc' ? -1 : 1);
  const get = (r) => {
    switch (field) {
      case 'Title': return (r.Title || '').toLowerCase();
      case 'AssignedTo': return (r.AssignedTo || '').toLowerCase();
      case 'Status': return (r.Status || '').toLowerCase();
      case 'Created': default: return r.Created ? new Date(r.Created).getTime() : 0;
    }
  };
  return rows.slice().sort((a,b) => (get(a) > get(b) ? 1 : (get(a) < get(b) ? -1 : 0)) * mul);
}
function sortTasks(rows, key) {
  const [field, dir] = key.split('_'); const mul = (dir === 'desc' ? -1 : 1);
  const get = (r) => {
    switch (field) {
      case 'AssignedTo': return (r.AssignedTo || '').toLowerCase();
      case 'Status': return (r.Status || '').toLowerCase();
      case 'DueDate': default: return r.DueDate ? new Date(r.DueDate).getTime() : Number.MAX_SAFE_INTEGER;
    }
  };
  return rows.slice().sort((a,b) => (get(a) > get(b) ? 1 : (get(a) < get(b) ? -1 : 0)) * mul);
}
function filterDocRow(row) {
  const f = ui.filters;
  const text = (f.search || '').toLowerCase();
  const passText = !text || (row.Title?.toLowerCase().includes(text) || (row.AssignedTo||'').toLowerCase().includes(text));
  const passCat = !f.category || row.Category === f.category;
  const passStatus = !f.status || row.Status === f.status;
  const passAssignee = !f.assignee || row.AssignedTo === f.assignee;
  return passText && passCat && passStatus && passAssignee;
}
function fillSelectFromSet(selectId, valuesSet, placeholder) {
  const el = document.getElementById(selectId);
  const prev = el.value;
  const options = [`<option value="">${placeholder}</option>`]
    .concat(Array.from(valuesSet).sort((a,b)=>String(a).localeCompare(String(b))).map(v => `<option value="${String(v)}">${String(v)}</option>`));
  el.innerHTML = options.join('');
  if (valuesSet.has(prev)) el.value = prev;
}

    if (btn.dataset.action === "loadAtt") {
      const id = btn.dataset.id;
      const email = (window.__state.colEmailsTracking || []).find(e => String(e.id) === String(id));
      if (email) {
        const container = document.getElementById(`attContainer_${CSS.escape(id)}`);
        if (container) { container.innerHTML = "<p class='muted'>Loading attachments…</p>"; await loadEmailAttachmentsInto(container, email); }
      }
    }

    /* Document attachment actions */
    if (btn.dataset.action === "openFile") {
      const url = btn.dataset.url; if (url) window.open(url, "_blank");
    }
    if (btn.dataset.action === "copyFile") {
      const url = btn.dataset.url;
      if (url) {
        try { await navigator.clipboard.writeText(url); showToast("Link copied.", "success"); }
        catch { showToast("Copy failed.", "error"); }
      }
    }
  });

  /* Telemetry */
  document.getElementById("btnTelemetryRefresh").onclick = renderTelemetry;
  document.getElementById("btnTelemetrySend").onclick    = sendTelemetryToFlow;
}

/* Startup */
document.addEventListener("DOMContentLoaded", async () => { wireEvents(); await refreshAll(); });
